МІНІ-РОУТЕР СВОЇМИ РУКАМИ

const routes = { "/": Home, "/about": About };

function Router() {
  const [path, setPath] = useState(location.pathname);

  useEffect(() => {
    const onPop = () => setPath(location.pathname);
    window.addEventListener("popstate", onPop);
    return () => window.removeEventListener("popstate", onPop);
  }, []);

  const navigate = (to) => {
    history.pushState({}, "", to);
    setPath(to);
  };

  const Page = routes[path] ?? NotFound;
  return <Page navigate={navigate} />;
}

CREATEBROWSERROUTER

Роутер створюється один раз — на рівні модуля, не в компоненті

import { createBrowserRouter, createRoutesFromElements,
         Route } from 'react-router';
import { RouterProvider } from 'react-router/dom';
import { Layout } from './Layout';
import { Home } from './pages/Home';
import { Lesson1 } from './pages/Lesson1';
import { Articles } from './pages/Articles';

// поза компонентом: інакше новий роутер на кожен рендер
const router = createBrowserRouter(
  createRoutesFromElements(
    <Route path="/" element={<Layout />}>
      <Route index element={<Home />} />
      <Route path="lesson1" element={<Lesson1 />} />
      <Route path="articles/:id" element={<Articles />} />
    </Route>
  )
);

function App() {
  return <RouterProvider router={router} />;
}

Місце, куди батьківський маршрут рендерить дочірній

const router = createBrowserRouter([
  {
    // it renders this element
    element: <Team />,

    // when the URL matches this segment
    path: "teams/:teamId",

    // with this data loaded before rendering
    loader: async ({ request, params }) => {
      return fetch(
        `/fake/api/teams/${params.teamId}.json`,
        { signal: request.signal }
      );
    },

    // performing this mutation when data is submitted to it
    action: async ({ request, params }) => {
      const formData = await request.formData();
      return updateTeam(params.teamId, formData);
    },

    // and rendering this element in case something went wrong
    errorElement: <TeamError />,
  },
]);

import { useEffect, useState } from "react";
import { useParams, useNavigate } from "react-router";

const Articles = () => {
  const { id } = useParams();
  const navigate = useNavigate();
  const [data, setData] = useState(null);

  useEffect(() => {
    const controller = new AbortController();
    fetch(`https://jsonplaceholder.typicode.com/posts/${id}`,
      { signal: controller.signal })
      .then(response => response.json())
      .then(setData)
      .catch(err => { if (err.name !== "AbortError") console.error(err); });
    return () => controller.abort();
  }, [id]); // ← ефект залежить від id

  return (
    <article>
      <button onClick={() => navigate(-1)}>Назад</button>
      <h1>{data?.title}</h1>
    </article>
  );
};

Обробка помилок завантаження lazy компонентів

ERROR BOUNDARY

Реалізація класового компонента

class ErrorBoundary extends React.Component {
  state = { hasError: false };

  static getDerivedStateFromError(error) {
    return { hasError: true };
  }

  componentDidCatch(error, errorInfo) {
    console.log('Error:', error);
    console.log('Error Info:', errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return <h1>Something went wrong.</h1>;
    }
    return this.props.children;
  }
}
