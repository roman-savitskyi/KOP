  title, price, oldPrice,
  badge, image, inStock, accent,
}) {
  return (
    <article>
      {badge && <span>{badge}</span>}
      <img src={image} alt={title} />
      <h3>{title}</h3>
      <p>
        {oldPrice && <s>{oldPrice} ₴</s>}

    </article>
  );
}

БЕМ: ІЗОЛЯЦІЯ ДОМОВЛЕНІСТЮ

Блок, Елемент, Модифікатор

.product-card                  ← Блок
.product-card__title           ← Блок__Елемент
.product-card__badge--sale     ← Блок__Елемент--Модифікатор
  • Імʼя блоку = імʼя компонента → конфлікти малоймовірні
  • Низька й однакова специфічність (один клас)
  • Але це домовленість: інструмент її не перевіряє

PLAIN CSS: ТЕМА

export function ThemeToggle() {
  const [theme, setTheme] = useState('light');

  useEffect(() => {
    document.documentElement.dataset.theme = theme; // <html data-theme="dark">
  }, [theme]);

  return (
    <button onClick={() => setTheme((t) => (t === 'light' ? 'dark' : 'light'))}>
      Тема
    </button>
  );
}
  • Змінні в :root і [data-theme="dark"] — слайд «CSS-змінні»
  • React лише перемикає атрибут, кольори перераховує браузер
  • Системна тема: @media (prefers-color-scheme: dark)

PLAIN CSS: ПІДСУМОК

ПЕРЕВАГИ

  • Просто, нічого не треба вивчати
  • Не потрібна конфігурація
  • Усі можливості CSS
  • 0 KB JS

НЕДОЛІКИ

  • Класи глобальні, а не локальні
  • Потрібні методології типу БЕМ
  • Перебивання стилів між компонентами
  • Важко знайти зайвий CSS

INLINE-СТИЛІ

Обʼєкт стилів у пропсі style

const cardStyle = {
  borderRadius: 12,          // число → 12px
  padding: '16px 20px',
  backgroundColor: '#fff',   // camelCase замість background-color
};

export function ProductCard({ title, accent }) {
  return (
    <article style={cardStyle}>
      <h3 style={{ color: accent }}>{title}</h3>
    </article>
  );
}
  • style приймає обʼєкт, а не рядок
  • {{ }} — це вираз JSX, всередині якого обʼєкт
  • Значення легко брати з пропсів і стану

ДЕ INLINE ЛАМАЄТЬСЯ

Спробуємо зробити ProductCard тільки inline

  • :hover, :disabled — неможливо
  • @media — неможливо
  • ::before, анімації @keyframes — неможливо
  • Найвища специфічність — перебити лише !important
  • Перевикористання — тільки копіюванням обʼєкта
// «hover» через стан — анти-патерн
const [hover, setHover] = useState(false);

<article
  style={{
    ...cardStyle,
    transform: hover ? 'translateY(-4px)' : 'none',
  }}
  onMouseEnter={() => setHover(true)}
  onMouseLeave={() => setHover(false)}
>

ПРЕПРОЦЕСОРИ

CSS препроцесор — це надбудова над CSS, яка додає додаткові можливості за допомогою синтаксичних конструкцій

  • SCSS — синтаксис як у CSS
  • Sass — без дужок і крапок з комою
  • Less — колись основа Bootstrap 3
  • Stylus

Компілюються у звичайний CSS під час збірки — браузер про них не знає

SCSS

npm i -D sass-embedded
$radius: 12px;
$shadow: 0 8px 24px rgba(0, 0, 0, .15);

@mixin lift($dy: -4px) {
  transition: transform .2s, box-shadow .2s;
  &:hover { transform: translateY($dy); box-shadow: $shadow; }
}

.card {
  border-radius: $radius;
  @include lift;

  .title { font-size: 18px; }

  @media (max-width: 640px) { padding: 12px; }
}

Змінні, міксини, вкладеність. .module.scss = SCSS + CSS Modules

@LAYER: КЕРУВАННЯ КАСКАДОМ

Хто перемагає — визначає порядок шарів, а не специфічність

@layer reset, base, components, utilities;

@layer reset {
  * { margin: 0; box-sizing: border-box; }
}
@layer components {
  .card .title { color: var(--color-text); }  /* 0-0-2-0 */
}
@layer utilities {
  .text-accent { color: var(--accent); }      /* 0-0-1-0 — перемагає! */
}
  • Порядок шарів важливіший за специфічність
  • Стилі бібліотек можна покласти в нижній шар — і легко перебити
  • Саме так влаштований Tailwind CSS v4

КОЛИ SASS ЩЕ ПОТРІБЕН

  • Міксини з параметрами
  • Цикли @each / @for — генерація набору класів
  • Функції та модулі (@use)
  • Існуючі проєкти, кастомізація Bootstrap

CSS-IN-JS: THEMEPROVIDER

import styled, { ThemeProvider } from 'styled-components';

const light = { bg: '#ffffff', text: '#1f2937' };
const dark = { bg: '#111827', text: '#f9fafb' };

const Card = styled.article`
  background: ${({ theme }) => theme.bg};
  color: ${({ theme }) => theme.text};
`;

export function App() {
  const [isDark, setIsDark] = useState(false);
  return (
    <ThemeProvider theme={isDark ? dark : light}>
      <ProductGrid />
    </ThemeProvider>
  );
}
  • Під капотом — React Context (наступна лекція)
  • Зміна теми → перерендер і перегенерація стилів у JS
  • З CSS-змінними той самий результат дає браузер, без JS

CSS-IN-JS: ПІДСУМОК

ПЕРЕВАГИ

  • Контроль CSS з JS
  • Відсутність конфлікту імен
  • Стилі залежать від пропсів
  • Стилі видаляються разом із компонентом

НЕДОЛІКИ

  • Швидкодія: стилі генеруються під час рендера
  • Збільшення кількості JS (бібліотека в бандлі)
  • Синтаксис: CSS у рядку, потрібен плагін для підсвітки
  • Складніший SSR

CSS-IN-JS У 2026 РОЦІ

  • styled-components з 2025 року в режимі підтримки (maintenance): лише виправлення
  • Emotion — та сама runtime-модель і ті самі обмеження
  • Runtime CSS-in-JS не працює з React Server Components
  • Zero-runtime: vanilla-extract, Panda CSS, Linaria — пишемо в JS/TS, на виході статичний CSS
// ProductCard.css.ts (vanilla-extract)
import { style } from '@vanilla-extract/css';

export const card = style({
  borderRadius: 12,
  ':hover': {
    transform: 'translateY(-4px)',
  },
});
// → звичайний .css файл під час збірки

Знати — щоб підтримувати існуючі проєкти. Для нового — не обирати runtime CSS-in-JS

TAILWIND CSS: UTILITY-FIRST

Не пишемо CSS — збираємо стилі з готових класів

export function ProductCard({ title, price, oldPrice, inStock }) {
  return (
    <article className="relative rounded-xl bg-white p-4 shadow-sm transition
                        hover:-translate-y-1 hover:shadow-lg">
      <span className="absolute left-3 top-3 rounded-full bg-red-600 px-2
                       text-xs text-white">Знижка</span>
      <h3 className="mt-2 text-lg font-semibold">{title}</h3>
      <p className="mt-1">
        {oldPrice && <s className="mr-2 text-gray-400">{oldPrice} ₴</s>}
        <strong className="text-xl">{price} ₴</strong>
      </p>
      <button className="mt-4 w-full rounded-lg bg-blue-600 py-2 text-white
                         disabled:cursor-not-allowed disabled:opacity-50"
              disabled={!inStock}>Купити</button>
    </article>
  );
}

Один клас = одна властивість: p-4 → padding: 1rem

UI-БІБЛІОТЕКИ: MUI

npm i @mui/material @emotion/react @emotion/styled
import {
  Card, CardContent, CardActions, Typography, Button, Chip,
} from '@mui/material';

export function ProductCard({ title, price, badge, inStock }) {
  return (
    <Card sx={{ borderRadius: 3, '&:hover': { boxShadow: 6 } }}>
      <CardContent>
        {badge && <Chip label={badge} color={badge === 'sale' ? 'error' : 'success'} />}
        <Typography variant="h6">{title}</Typography>
        <Typography variant="h5" color="primary">{price} ₴</Typography>
      </CardContent>
      <CardActions>
        <Button variant="contained" fullWidth disabled={!inStock}>Купити</Button>
      </CardActions>
    </Card>
  );
}
  • sx — стилі через проп; тема — createTheme + ThemeProvider
  • Під капотом Emotion — runtime CSS-in-JS з його обмеженнями

UI-БІБЛІОТЕКИ: ЯК ОБРАТИ

  • Адмінка, внутрішній інструмент, треба швидко → MUI / Mantine / Ant Design
  • Власний дизайн, потрібен повний контроль → shadcn/ui або headless + свої стилі

КОЛИ ЗАВАЖАЄ

  • Дизайн сильно відрізняється → боротьба з перевизначенням стилів
  • Розмір бандлу
  • Мажорні оновлення ламають API
