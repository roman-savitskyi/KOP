КОМПОЗИЦІЯ ЗАМІСТЬ CONTEXT

Часто prop drilling вирішується без контексту

function App() {
  const [cart, setCart] = useState([]);
  const add = (product) => setCart((c) => [...c, product]);

  return (
    <Layout header={<Header count={cart.length} />}>
      <ProductGrid onAdd={add} />
    </Layout>
  );
}

// Layout більше не знає про кошик
function Layout({ header, children }) {
  return (
    <>
      {header}
      <main>{children}</main>
    </>
  );
}
  • Проміжні компоненти отримують готові елементи, а не дані
  • Документація React: спершу спробуйте композицію, потім Context

КОЛИ ВСЕ Ж CONTEXT

  • Дані потрібні багатьом компонентам на різній глибині
  • Дані змінюються рідко
  • Тема, мова інтерфейсу, авторизований користувач
  • Бібліотеки теж на ньому: роутер (useNavigate), ThemeProvider зі styled-components

Context — це механізм доставки даних, а не менеджер стану

THEMEPROVIDER

Тема для магазину

// theme/ThemeContext.jsx
import { createContext, useState } from 'react';

export const ThemeContext = createContext(null);

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light');
  const toggle = () => setTheme((t) => (t === 'light' ? 'dark' : 'light'));

  return (
    <ThemeContext value={{ theme, toggle }}>
      <div data-theme={theme}>{children}</div>
    </ThemeContext>
  );
}
  • React 19: контекст сам є провайдером. До 19 — <ThemeContext.Provider>
  • data-theme — CSS-змінні теми з минулої лекції

КАСТОМНИЙ ХУК ДЛЯ КОНТЕКСТУ

export function useTheme() {
  const ctx = useContext(ThemeContext);
  if (ctx === null) {
    throw new Error('useTheme() потрібно викликати всередині <ThemeProvider>');
  }
  return ctx;
}

// ThemeToggle.jsx
export function ThemeToggle() {
  const { theme, toggle } = useTheme();
  return <button onClick={toggle}>{theme === 'light' ? '🌙' : '☀️'}</button>;
}
  • Компоненти знають лише про хук, а не про сам контекст
  • Зрозуміла помилка замість Cannot read properties of null
  • Сам ThemeContext можна не експортувати

ПРОБЛЕМА 1: НОВИЙ ОБʼЄКТ VALUE

// ❌ новий обʼєкт на кожен рендер
function CartProvider({ children }) {
  const [items, setItems] = useState([]);
  const add = (p) =>
    setItems((prev) => [...prev, p]);

  return (
    <CartContext value={{ items, add }}>
      {children}
    </CartContext>
  );
}
// ✅ стабільні посилання
function CartProvider({ children }) {
  const [items, setItems] = useState([]);
  const add = useCallback((p) =>
    setItems((prev) => [...prev, p]), []);

  const value = useMemo(
    () => ({ items, add }), [items, add]);

  return (
    <CartContext value={value}>
      {children}
    </CartContext>
  );
}
  • React порівнює value через Object.is
  • Ререндер батька провайдера → новий {} → ререндер усіх споживачів

ПРОБЛЕМА 2: РОЗДІЛИТИ STATE І ACTIONS

const CartStateContext = createContext(null);
const CartActionsContext = createContext(null);

function CartProvider({ children }) {
  const [items, setItems] = useState([]);

  const actions = useMemo(() => ({
    add: (p) => setItems((prev) => [...prev, p]),
    remove: (id) => setItems((prev) => prev.filter((i) => i.id !== id)),
  }), []);

  return (
    <CartActionsContext value={actions}>
      <CartStateContext value={items}>{children}</CartStateContext>
    </CartActionsContext>
  );
}

КОШИК: REDUCER

Context + useReducer = «міні-Redux»

// cart/cartReducer.js
export function cartReducer(items, action) {
  switch (action.type) {
    case 'added': {
      const existing = items.find((i) => i.id === action.product.id);
      if (existing) {
        return items.map((i) => (i.id === existing.id ? { ...i, qty: i.qty + 1 } : i));
      }
      return [...items, { ...action.product, qty: 1 }];
    }
    case 'removed':
      return items.filter((i) => i.id !== action.id);
    case 'cleared':
      return [];
    default:
      throw new Error(`Невідома дія: ${action.type}`);
  }
}

Уся логіка кошика — в одній чистій функції. Легко тестувати без React

КОШИК: PROVIDER

// cart/CartProvider.jsx
const CartContext = createContext(null);
const CartDispatchContext = createContext(null);

export function CartProvider({ children }) {
  const [items, dispatch] = useReducer(cartReducer, []);

  return (
    <CartDispatchContext value={dispatch}>
      <CartContext value={items}>{children}</CartContext>
    </CartDispatchContext>
  );
}

export const useCart = () => use(CartContext);
export const useCartDispatch = () => use(CartDispatchContext);

dispatch стабільний між рендерами — useMemo не потрібен

import { atom, useAtomValue, useSetAtom } from 'jotai';

const itemsAtom = atom([]);
const countAtom = atom((get) => get(itemsAtom).length); // похідний

СИГНАЛИ

Реактивність на рівні значення, а не компонента

  • Сигнал = значення + список тих, хто його читає
  • Читання → автоматична підписка
  • Зміна → оновлюються лише залежні обчислення і шматки DOM
  • Fine-grained reactivity
React:
  setState → ререндер компонента
           → ререндер дітей
           → порівняння Virtual DOM
           → оновлення DOM

Сигнали:
  count.value++ → оновлення
                  одного текстового
                  вузла в DOM

Тренд останніх років: Vue, Solid, Svelte 5, Preact, Qwik

ЩО ОБРАТИ?

  • Тема, мова, користувач → Context
  • Кошик, фільтри в середньому застосунку → Zustand (React) / Pinia (Vue)
  • Великий застосунок, команда, складні потоки даних → Redux Toolkit
  • Багато дрібних незалежних шматків стану → Jotai
  • Дані з сервера → не в store: TanStack Query, RTK Query, loader

Головне — не тягнути весь стан у глобальне сховище
