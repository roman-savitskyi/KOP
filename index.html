Базові форми

Керовані компоненти

import { useState } from 'react';

export function Form() {
  const [value, setValue] = useState(0);

  const handleSubmit = (event) => {
    console.log('submit value', value);
    event.preventDefault();
  }

  const handleChange = (event) => {
    setValue(event.target.value);
  }

  return (<form onSubmit={handleSubmit}>
    <label>
      Ім'я:
      <input type="text" value={value} onChange={handleChange} />
    </label>
    <input type="submit" value="Надіслати" />
  </form>);
}

Некеровані компоненти

import { useRef } from 'react';

export function FormUncontrol() {
  const handleSubmit = (event) => {
    console.log('submit value', input.current.value);
    event.preventDefault();
  }

  const input = useRef();

  return (<form onSubmit={handleSubmit}>
    <label>
      Ім'я:
      <input type="text" ref={input} />
    </label>
    <input type="submit" value="Надіслати" />
  </form>);
}

import React from 'react';

export function ReactHookForm() {
  const { register, handleSubmit, formState: { errors } } = useForm();
  const onSubmit = data => console.log(data);
  console.log(errors);

  return (
    <form onSubmit={handleSubmit(onSubmit)}>

}

Стан форми

import { useForm } from 'react-hook-form';

export function App() {
  const { register, formState } = useForm();
  const { isDirty, isValid, isSubmitting, touchedFields,
          dirtyFields, errors } = formState;

  return (
    <form>
      <input {...register("firstName")} />
      <button disabled={!isDirty || !isValid}>
        Надіслати
      </button>
    </form>
  );
}

Властивості стану форми

  • isDirty - чи змінювалась форма
  • isValid - чи валідна форма
  • isSubmitting - чи відправляється форма
  • isSubmitted - чи була відправлена
  • touchedFields - поля, яких торкались
  • dirtyFields - змінені поля

import { useForm } from 'react-hook-form';

export function App() {
  const { register, handleSubmit, reset } = useForm({
    defaultValues: { firstName: "" }
  });

  const onSubmit = data => {
    console.log(data);
    reset(); // Скидання до defaultValues

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input {...register("firstName")} />
      <button type="submit">Надіслати</button>
      <button type="button" onClick={() => reset()}>

      </button>

}

Динамічні форми

useFieldArray

import { useForm, useFieldArray } from 'react-hook-form';

export function App() {
  const { register, control } = useForm();
  const { fields, append, remove } = useFieldArray({
    control,
    name: "users"
  });

  return (
    <form>
      {fields.map((field, index) => (
        <input key={field.id} {...register(`users.${index}.name`)} />
      ))}
    </form>
  );
}

Динамічні поля

Додавання та видалення

export function DynamicForm() {
  const { register, control } = useForm({
    defaultValues: { items: [{ name: '' }] }
  });

  const { fields, append, remove } = useFieldArray({
    control, name: "items"
  });

  return (
    <form>
      {fields.map((field, index) => (
        <div key={field.id}>
          <input {...register(`items.${index}.name`)} />
          <button onClick={() => remove(index)}>Видалити</button>
        </div>
      ))}

      <button onClick={() => append({ name: '' })}>
        Додати поле
      </button>
    </form>
  );
}

USEFIELDARRY методи

  • fields - масив полів
  • append(data) - додати в кінець
  • prepend(data) - додати на початок
  • remove(index) - видалити за індексом
  • insert(index, data) - вставити
  • swap(indexA, indexB) - поміняти місцями

YUP RESOLVER

Інтеграція з React Hook Form

import { useForm } from 'react-hook-form';
import { yupResolver } from '@hookform/resolvers/yup';
import * as yup from 'yup';

const schema = yup.object({
  firstName: yup.string().required("Ім'я обов'язкове"),
}).required();

export function App() {
  const { register, handleSubmit, formState: { errors } } = useForm({
    resolver: yupResolver(schema)
  });

  return (
    <form onSubmit={handleSubmit(data => console.log(data))}>
      <input {...register("firstName")} />
      <p>{errors.firstName?.message}</p>
      <input type="submit" />
    </form>
  );
}

YUP МЕТОДИ ВАЛІДАЦІЇ

  • string() - рядок (min, max, email, url, matches)

  • boolean() - логічне значення
  • date() - дата (min, max)
  • array() - масив (min, max, of)
  • object() - об'єкт (shape)
